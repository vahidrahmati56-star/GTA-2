function makeCanvasTexture(w, h, drawFn, repeat=1){
  const c = document.createElement('canvas');
  c.width = w; c.height = h;
  const ctx = c.getContext('2d');
  drawFn(ctx, w, h);
  const tex = new THREE.CanvasTexture(c);
  tex.wrapS = tex.wrapT = THREE.RepeatWrapping;
  tex.repeat.set(repeat, repeat);
  tex.anisotropy = renderer.capabilities.getMaxAnisotropy();
  tex.colorSpace = THREE.SRGBColorSpace;
  return tex;
}

(function addGrassTufts(){
  const bladeGeo = new THREE.PlaneGeometry(0.35, 0.5);
  const bladeMat = new THREE.MeshStandardMaterial({
    color: 0x5a9540, side: THREE.DoubleSide,
    transparent:true, alphaTest:0.5, roughness:1
  });
  // تکسچر چمن ساده
  const grassBladeTex = makeCanvasTexture(64, 64, (ctx,w,h)=>{
    ctx.clearRect(0,0,w,h);
    const grad = ctx.createLinearGradient(0, h, 0, 0);
    grad.addColorStop(0, '#3a6a28');
    grad.addColorStop(1, '#6aa848');
    ctx.fillStyle = grad;
    for(let i=0;i<18;i++){
      const x = 6 + Math.random()*52;
      const hh = 30 + Math.random()*28;
      const ww = 2 + Math.random()*2;
      ctx.beginPath();
      ctx.moveTo(x, h);
      ctx.quadraticCurveTo(x + (Math.random()-0.5)*8, h - hh*0.5, x + (Math.random()-0.5)*10, h - hh);
      ctx.lineTo(x + ww, h - hh);
      ctx.quadraticCurveTo(x + ww + (Math.random()-0.5)*8, h - hh*0.5, x + ww, h);
      ctx.fill();
    }
  });
  bladeMat.map = grassBladeTex;
  bladeMat.alphaMap = grassBladeTex;
  bladeMat.alphaTest = 0.4;

  const N = 3000;
  const inst = new THREE.InstancedMesh(bladeGeo, bladeMat, N);
  inst.receiveShadow = true;
  const dummy = new THREE.Object3D();
  for(let i=0;i<N;i++){
    let x, z;
    // پراکنده روی زمین ولی نه روی جاده‌ها
    do {
      x = (Math.random()-0.5)*180;
      z = (Math.random()-0.5)*180;
    } while(Math.abs(x) < 8 || Math.abs(z) < 8);
    dummy.position.set(x, 0.25, z);
    dummy.rotation.y = Math.random()*Math.PI;
    const s = 0.7 + Math.random()*0.9;
    dummy.scale.set(s, s*(0.8+Math.random()*0.6), s);
    dummy.updateMatrix();
    inst.setMatrixAt(i, dummy.matrix);
  }
  scene.add(inst);
})();

    b.castShadow = true; g.add(b);
  }
  // برگ‌ها در چند لایه با رنگ‌های متنوع

    const foliage = new THREE.Mesh(
      new THREE.IcosahedronGeometry(1.5 - i*0.12, 1),
      new THREE.MeshStandardMaterial({color: c, roughness:1, flatShading:true})
    );
    const a = (i/5)*Math.PI*2;
    const r = i === 0 ? 0 : 0.9;
    foliage.position.set(Math.cos(a)*r, 4.2 + Math.sin(i*1.7)*0.4, Math.sin(a)*r);
    foliage.castShadow = true;
    g.add(foliage);
  }
  const top = new THREE.Mesh(new THREE.IcosahedronGeometry(1.4, 1),
    new THREE.MeshStandardMaterial({color:0x3a7a2a, roughness:1, flatShading:true}));
  top.position.y = 5.0; top.castShadow = true; g.add(top);

  g.position.set(x, 0, z); g.scale.setScalar(scale);
  scene.add(g); return g;
}

      new THREE.IcosahedronGeometry(1.2 - i*0.1, 1),
      new THREE.MeshStandardMaterial({color: leafColors[i%3], roughness:1, flatShading:true})
    );

    f.position.set(Math.cos(a)*r, 3.4 + Math.sin(i*2)*0.3, Math.sin(a)*r);

  }

    new THREE.MeshStandardMaterial({color:0x3a8a30, roughness:1, flatShading:true}));
  top.position.y = 4.3; top.castShadow = true; g.add(top);

    color:0xb02222, roughness:0.35, metalness:0.05,
    emissive:0x2a0000, emissiveIntensity:0.3
  });
  for(let i=0;i<20;i++){
    const f = new THREE.Mesh(new THREE.SphereGeometry(0.17, 12, 12), fruitMat);
    const a = Math.random()*Math.PI*2;
    const b = Math.random()*Math.PI;

// ساختمان‌های واقع‌گرایانه با تکسچر گِلی و چوبی و پنجره
function makeBuilding(x, z, w, h, d, tint=0xffffff, roofColor=0x8a3a2a, style='adobe'){
  const g = new THREE.Group();
  const wallTex = style === 'adobe' ? adobeTex.clone() : woodTex.clone();
  wallTex.needsUpdate = true;
  wallTex.repeat.set(Math.max(1, w/3), Math.max(1, h/3));
  const bodyMat = new THREE.MeshStandardMaterial({
    map: wallTex, roughness:0.92, metalness:0.0, color: tint
  });
  const body = new THREE.Mesh(new THREE.BoxGeometry(w, h, d), bodyMat);
  body.position.y = h/2; body.castShadow = true; body.receiveShadow = true;
  g.add(body);

  // سقف شیروانی
  const roofGeo = new THREE.ConeGeometry(Math.max(w,d)*0.78, 1.8, 4);
  const roofMat = new THREE.MeshStandardMaterial({color: roofColor, roughness:0.85});
  const roof = new THREE.Mesh(roofGeo, roofMat);
  roof.position.y = h + 0.9; roof.rotation.y = Math.PI/4;
  roof.castShadow = true; roof.receiveShadow = true;
  g.add(roof);

  // درِ چوبی
  const doorMat = new THREE.MeshStandardMaterial({map: woodTex, roughness:0.85, color:0x5a3a20});
  const door = new THREE.Mesh(new THREE.BoxGeometry(1.2, 2.2, 0.12), doorMat);
  door.position.set(0, 1.1, d/2 + 0.06); door.castShadow = true; g.add(door);
  // دستگیره برنجی
  const knob = new THREE.Mesh(new THREE.SphereGeometry(0.07, 10, 10),
    new THREE.MeshStandardMaterial({color:0xc8a040, roughness:0.3, metalness:0.85}));
  knob.position.set(0.45, 1.1, d/2 + 0.14); g.add(knob);

  // پنجره‌ها
  const glassMat = new THREE.MeshPhysicalMaterial({
    color:0x88bbdd, roughness:0.05, metalness:0, transmission:0.7,
    transparent:true, opacity:0.85, thickness:0.02
  });
  const frameMat = new THREE.MeshStandardMaterial({color:0x3a2818, roughness:0.8});
  function addWindow(px, py, pz, ry){
    const wg = new THREE.Group();
    const glass = new THREE.Mesh(new THREE.PlaneGeometry(1.0, 1.2), glassMat);
    wg.add(glass);
    const fr = new THREE.Mesh(new THREE.BoxGeometry(1.15, 1.35, 0.06), frameMat);
    fr.position.z = -0.04; wg.add(fr);
    // میله‌های صلیبی
    const cross1 = new THREE.Mesh(new THREE.BoxGeometry(0.06, 1.2, 0.05), frameMat);
    wg.add(cross1);
    const cross2 = new THREE.Mesh(new THREE.BoxGeometry(1.0, 0.06, 0.05), frameMat);
    wg.add(cross2);
    wg.position.set(px, py, pz); wg.rotation.y = ry;
    g.add(wg);
  }
  const halfW = w/2 - 0.9, halfD = d/2 + 0.05;
  addWindow(-halfW, 2.4, halfD, 0);
  addWindow( halfW, 2.4, halfD, 0);
  if(d > 4){ addWindow(halfD, 2.4, 0, Math.PI/2); }
  if(d > 4){ addWindow(-halfD, 2.4, 0, -Math.PI/2); }

  g.position.set(x, 0, z);
  scene.add(g); return g;
}

function makeCharacter(opts={}){
  const {shirt=0x3a5fc8, pants=0x2a2a3a, skin=0xe8b088, glasses=true, hairColor=0x1a1a1a, gender='m'} = opts;
  const c = new THREE.Group();

  // پوست با تکسچر دقیق
  const skinTex = makeCanvasTexture(128, 128, (ctx,w,h)=>{
    ctx.fillStyle = '#' + skin.toString(16).padStart(6,'0');
    ctx.fillRect(0,0,w,h);
    for(let i=0;i<600;i++){
      const a = 0.02 + Math.random()*0.06;
      ctx.fillStyle = `rgba(${180+Math.random()*60},${130+Math.random()*40},${90+Math.random()*40},${a})`;
      ctx.fillRect(Math.random()*w, Math.random()*h, 1+Math.random()*2, 1+Math.random()*2);
    }
  }, 1);

  const skinM = new THREE.MeshStandardMaterial({
    map: skinTex, color: skin, roughness: 0.72, metalness: 0.0
  });
  const shirtM = new THREE.MeshStandardMaterial({color:shirt, roughness:0.88, metalness:0.02});
  const pantsM = new THREE.MeshStandardMaterial({color:pants, roughness:0.92});
  const darkM  = new THREE.MeshStandardMaterial({color:0x0a0a0a, metalness:0.6, roughness:0.3});
  const lensM  = new THREE.MeshPhysicalMaterial({
    color:0xaaddff, transparent:true, opacity:0.25, roughness:0.05,
    metalness:0.0, transmission:0.6, thickness:0.02, ior:1.45
  });
  const whiteM = new THREE.MeshStandardMaterial({color:0xf8f8f8, roughness:0.4});
  const pupilM = new THREE.MeshStandardMaterial({color:0x0a0a10, roughness:0.2});
  const hairM  = new THREE.MeshStandardMaterial({color:hairColor, roughness:0.85});

  // === بدن / نیم‌تنه ===
  const torso = new THREE.Mesh(new THREE.CapsuleGeometry(0.28, 0.55, 8, 16), shirtM);
  torso.position.y = 1.18; torso.castShadow = true; torso.receiveShadow = true; c.add(torso);

  // شانه‌ها کمی پهن‌تر
  const shoulders = new THREE.Mesh(new THREE.BoxGeometry(0.72, 0.14, 0.36),
    new THREE.MeshStandardMaterial({color:shirt, roughness:0.9}));
  shoulders.position.y = 1.50; shoulders.castShadow = true; c.add(shoulders);

  // === گردن ===
  const neck = new THREE.Mesh(new THREE.CylinderGeometry(0.09, 0.11, 0.12, 10), skinM);
  neck.position.y = 1.62; neck.castShadow = true; c.add(neck);

  // === سر (بیضی طبیعی) ===
  const head = new THREE.Mesh(new THREE.SphereGeometry(0.22, 28, 28), skinM);
  head.position.y = 1.79; head.scale.set(1, 1.13, 1.02);
  head.castShadow = true; head.receiveShadow = true; c.add(head);

  // === مو با حجم طبیعی ===
  const hair = new THREE.Mesh(
    new THREE.SphereGeometry(0.238, 24, 24, 0, Math.PI*2, 0, Math.PI*0.62),
    hairM
  );
  hair.position.y = 1.815; hair.scale.set(1.02, 1.15, 1.04);
  hair.castShadow = true; c.add(hair);
  // چتری کوچک جلو
  const fringe = new THREE.Mesh(new THREE.BoxGeometry(0.36, 0.06, 0.10), hairM);
  fringe.position.set(0, 1.94, -0.16); fringe.rotation.x = 0.25; c.add(fringe);

  // === گوش‌ها ===
  [-1,1].forEach(s=>{
    const ear = new THREE.Mesh(new THREE.SphereGeometry(0.052, 10, 10), skinM);
    ear.position.set(s*0.215, 1.79, 0.01); ear.scale.set(0.7, 1, 0.8); c.add(ear);
  });

  // === ابروها ===
  [-0.082, 0.082].forEach(x=>{
    const brow = new THREE.Mesh(new THREE.BoxGeometry(0.09, 0.018, 0.03),
      new THREE.MeshStandardMaterial({color:hairColor, roughness:0.9}));
    brow.position.set(x, 1.878, -0.19); brow.rotation.z = x > 0 ? -0.12 : 0.12;
    c.add(brow);
  });

  // === چشم‌ها (کره چشم + عنبیه + مردمک) ===
  [-0.078, 0.078].forEach(x=>{
    const eye = new THREE.Mesh(new THREE.SphereGeometry(0.036, 16, 16), whiteM);
    eye.position.set(x, 1.838, -0.192); eye.scale.set(1, 0.9, 0.7); c.add(eye);
    const iris = new THREE.Mesh(new THREE.SphereGeometry(0.020, 12, 12),
      new THREE.MeshStandardMaterial({color:0x4a3520, roughness:0.15, metalness:0.1}));
    iris.position.set(x, 1.838, -0.220); iris.scale.set(1, 1, 0.55); c.add(iris);
    const pupil = new THREE.Mesh(new THREE.SphereGeometry(0.010, 10, 10), pupilM);
    pupil.position.set(x, 1.838, -0.227); pupil.scale.set(1, 1, 0.5); c.add(pupil);
    // برق چشم (highlight)
    const shine = new THREE.Mesh(new THREE.SphereGeometry(0.005, 8, 8),
      new THREE.MeshBasicMaterial({color:0xffffff}));
    shine.position.set(x + 0.006, 1.845, -0.228); c.add(shine);
  });

  // === بینی ===
  const noseBase = new THREE.Mesh(new THREE.ConeGeometry(0.036, 0.10, 8), skinM);
  noseBase.rotation.x = Math.PI/2; noseBase.position.set(0, 1.79, -0.215); c.add(noseBase);
  const noseTip = new THREE.Mesh(new THREE.SphereGeometry(0.028, 12, 12), skinM);
  noseTip.position.set(0, 1.775, -0.235); c.add(noseTip);

  // === لب ===
  const upperLip = new THREE.Mesh(new THREE.BoxGeometry(0.10, 0.012, 0.02),
    new THREE.MeshStandardMaterial({color:0xa04040, roughness:0.6}));
  upperLip.position.set(0, 1.735, -0.218); c.add(upperLip);
  const lowerLip = new THREE.Mesh(new THREE.BoxGeometry(0.11, 0.014, 0.02),
    new THREE.MeshStandardMaterial({color:0xa84848, roughness:0.6}));
  lowerLip.position.set(0, 1.722, -0.219); c.add(lowerLip);

    const frameMat = new THREE.MeshStandardMaterial({color:0x0a0a0a, metalness:0.6, roughness:0.3});
    [-0.078, 0.078].forEach(x=>{
      const ring = new THREE.Mesh(new THREE.TorusGeometry(0.058, 0.008, 10, 28), frameMat);
      ring.position.set(x, 1.838, -0.198); c.add(ring);
      const lens = new THREE.Mesh(new THREE.CircleGeometry(0.056, 28), lensM);
      lens.position.set(x, 1.838, -0.200); c.add(lens);
    });
    const bridge = new THREE.Mesh(new THREE.BoxGeometry(0.055, 0.008, 0.008), frameMat);
    bridge.position.set(0, 1.838, -0.198); c.add(bridge);
    [-1,1].forEach(s=>{
      const arm = new THREE.Mesh(new THREE.BoxGeometry(0.20, 0.008, 0.008), frameMat);
      arm.position.set(s*0.16, 1.838, -0.10); c.add(arm);
    });
  }

  // === بازوها (بازو + آرنج + ساعد + دست) ===
  function makeArm(side){
    const shoulder = new THREE.Group();
    shoulder.position.set(side*0.35, 1.45, 0);
    const upperArm = new THREE.Mesh(new THREE.CapsuleGeometry(0.072, 0.34, 6, 12), shirtM);
    upperArm.position.y = -0.24; upperArm.castShadow = true; shoulder.add(upperArm);
    const elbow = new THREE.Group();
    elbow.position.y = -0.48;
    const forearm = new THREE.Mesh(new THREE.CapsuleGeometry(0.062, 0.30, 6, 12), shirtM);
    forearm.position.y = -0.20; forearm.castShadow = true; elbow.add(forearm);
    // دست با انگشت‌های اشاره
    const hand = new THREE.Mesh(new THREE.BoxGeometry(0.12, 0.16, 0.07), skinM);
    hand.position.y = -0.42; hand.castShadow = true; elbow.add(hand);
    const thumb = new THREE.Mesh(new THREE.CapsuleGeometry(0.02, 0.06, 4, 6), skinM);
    thumb.position.set(side*0.06, -0.40, -0.04);
    thumb.rotation.z = side*0.6; elbow.add(thumb);
    // ۴ انگشت
    for(let i=0;i<4;i++){
      const fg = new THREE.Mesh(new THREE.CapsuleGeometry(0.018, 0.05, 4, 6), skinM);
      fg.position.set(-0.045 + i*0.03, -0.50, -0.02);
      fg.rotation.x = -0.15; elbow.add(fg);
    }
    shoulder.add(elbow);
    c.add(shoulder);
    return shoulder;
  }

  // === پاها (ران + ساق + کفش) ===
  function makeLeg(side){
    const hip = new THREE.Group();
    hip.position.set(side*0.13, 0.80, 0);
    const thigh = new THREE.Mesh(new THREE.CapsuleGeometry(0.10, 0.32, 6, 12), pantsM);
    thigh.position.y = -0.24; thigh.castShadow = true; hip.add(thigh);
    const knee = new THREE.Group();
    knee.position.y = -0.46;
    const shin = new THREE.Mesh(new THREE.CapsuleGeometry(0.085, 0.32, 6, 12), pantsM);
    shin.position.y = -0.22; shin.castShadow = true; knee.add(shin);
    // کفش چرمی
    const shoeMat = new THREE.MeshStandardMaterial({color:0x181818, roughness:0.55, metalness:0.05});
    const shoe = new THREE.Mesh(new THREE.BoxGeometry(0.16, 0.09, 0.28), shoeMat);
    shoe.position.set(0, -0.46, -0.05); shoe.castShadow = true; knee.add(shoe);
    const shoeTop = new THREE.Mesh(new THREE.BoxGeometry(0.14, 0.06, 0.16), shoeMat);
    shoeTop.position.set(0, -0.40, 0.05); knee.add(shoeTop);
    hip.add(knee);
    c.add(hip);
    return hip;
  }

  c.userData = {
    arms:{l:makeArm(-1), r:makeArm(1)},
    legs:{l:makeLeg(-1), r:makeLeg(1)},
    head
  };
  return c;
}

function joyStart(e){
  const t = e.changedTouches ? e.changedTouches[0] : e;
  if(joy.id !== null) return;
  joy.id = t.identifier !== undefined ? t.identifier : 'mouse';
  joy.active = true;
  const rect = joyEl.getBoundingClientRect();
  joy.cx = rect.left + rect.width/2;
  joy.cy = rect.top + rect.height/2;
  joyMove(e); e.preventDefault();
}
function joyMove(e){
  if(!joy.active) return;
  const touches = e.changedTouches || [e];
  for(const t of touches){
    const tid = t.identifier !== undefined ? t.identifier : 'mouse';
    if(tid !== joy.id) continue;
    const dx = t.clientX - joy.cx, dy = t.clientY - joy.cy;
    const maxR = 45, d = Math.hypot(dx, dy);
    const cl = Math.min(1, d/maxR), ang = Math.atan2(dy, dx);
    const kx = Math.cos(ang)*cl*maxR, ky = Math.sin(ang)*cl*maxR;
    joyKnob.style.transform = `translate(calc(-50% + ${kx}px), calc(-50% + ${ky}px))`;
    joy.dx = Math.cos(ang)*cl; joy.dy = Math.sin(ang)*cl;
  }
  e.preventDefault();
}
function joyEnd(e){
  if(!joy.active) return;
  const touches = e.changedTouches || [e];
  for(const t of touches){
    const tid = t.identifier !== undefined ? t.identifier : 'mouse';
    if(tid === joy.id){
      joy.active = false; joy.id = null; joy.dx = 0; joy.dy = 0;
      joyKnob.style.transform = 'translate(-50%,-50%)';
    }
  }
}

const dustParticles = (()=>{

  const pos = new Float32Array(N*3);

    pos[i*3+2] = -35 + (Math.random()-0.5)*30;
    vel[i*3]   = (Math.random()-0.5)*0.2;
    vel[i*3+1] = Math.random()*0.15 + 0.05;
    vel[i*3+2] = (Math.random()-0.5)*0.2;
  }
  geo.setAttribute('position', new THREE.BufferAttribute(pos, 3));
  const mat = new THREE.PointsMaterial({
    color:0xc8a878, size:0.22, transparent:true, opacity:0.55,
    depthWrite:false, sizeAttenuation:true
  });
  const p = new THREE.Points(geo, mat);
  scene.add(p);
  return {points:p, vel, geo};
})();
const dustEl = document.getElementById('dust');

function updateDust(dt){
  let inDust = false;
  for(const z of dustZones){
    const dx = player.pos.x - z.x, dz = player.pos.z - z.z;
    if(dx*dx + dz*dz < z.r*z.r){ inDust = true; break; }
  }
  if(inDust) player.dustT = Math.min(1, player.dustT + dt*2);
  else player.dustT = Math.max(0, player.dustT - dt*1.5);
  player.covering = player.dustT > 0.35;
  dustEl.classList.toggle('show', player.dustT > 0.35);

  // حرکت ذرات
  const posAttr = dustParticles.geo.attributes.position;
  const arr = posAttr.array;
  const vel = dustParticles.vel;
  for(let i=0;i<arr.length;i+=3){
    arr[i]   += vel[i]   * dt * 2;
    arr[i+1] += vel[i+1] * dt * 2;
    arr[i+2] += vel[i+2] * dt * 2;
    if(arr[i+1] > 5){ arr[i+1] = 0; }
    if(arr[i] < -65) arr[i] = -25;
    if(arr[i] > -25) arr[i] = -65;
    if(arr[i+2] < -55) arr[i+2] = -15;
    if(arr[i+2] > -15) arr[i+2] = -55;
  }
  posAttr.needsUpdate = true;
}

function updatePlayer(dt){
  if(state.dialogueOpen) return;
  const running = runHeld || keys['shift'];
  const speed = player.speed * (running ? 1.85 : 1);

  if(keys['a'] || keys['arrowleft']) player.rot += 2.4*dt;
  if(keys['d'] || keys['arrowright']) player.rot -= 2.4*dt;

  let mx = 0, my = 0;
  if(joy.active){ mx = joy.dx; my = -joy.dy; }
  if(keys['w'] || keys['arrowup']) my = 1;
  if(keys['s'] || keys['arrowdown']) my = -1;

  let moved = false;
  if(Math.abs(mx) > 0.05 || Math.abs(my) > 0.05){
    const forward = new THREE.Vector3(-Math.sin(player.rot), 0, -Math.cos(player.rot));
    const right   = new THREE.Vector3(Math.cos(player.rot), 0, -Math.sin(player.rot));
    const v = new THREE.Vector3().addScaledVector(forward, my).addScaledVector(right, mx);
    if(v.length() > 0.05){
      v.normalize();
      player.pos.addScaledVector(v, speed*dt);
      moved = true;
    }
  }

  player.pos.x = Math.max(-90, Math.min(90, player.pos.x));
  player.pos.z = Math.max(-90, Math.min(90, player.pos.z));
  player.running = running && moved;

  if(player.running && state.hasBasket && state.basketFruit > 0){
    player._dropAcc = (player._dropAcc || 0) + dt;
    if(player._dropAcc > 1.2){
      player._dropAcc = 0;
      state.basketFruit = Math.max(0, state.basketFruit - 1);
      showToast('🍎 یه انار افتاد! آروم‌تر برو'); updateHUD();
    }
  }

  // انیمیشن راه رفتن
  if(moved){
    player.walkPhase += dt * (player.running ? 14 : 9);
    const swing = Math.sin(player.walkPhase) * (player.running ? 0.85 : 0.5);
    player.legs.l.rotation.x =  swing;
    player.legs.r.rotation.x = -swing;
    // خم شدن آرنج
    if(!player.covering){
      player.arms.l.rotation.x = -swing * 0.6;
      player.arms.r.rotation.x =  swing * 0.6;
      player.arms.l.rotation.z = 0;
      player.arms.r.rotation.z = 0;
    }
    // tکان عمودی جزئی
    cameraBob += dt * (player.running ? 14 : 9);
  } else {
    player.walkPhase *= 0.85;
    player.legs.l.rotation.x *= 0.85;
    player.legs.r.rotation.x *= 0.85;
    if(!player.covering){
      player.arms.l.rotation.x *= 0.85;
      player.arms.r.rotation.x *= 0.85;
    }
  }

  // پوشاندن صورت
  const targetT = player.covering ? 1 : 0;
  player.coverT += (targetT - player.coverT) * Math.min(1, dt*6);
  const cT = player.coverT;
  player.arms.r.rotation.x = -cT * 2.6;
  player.arms.r.rotation.z =  cT * 0.6;
  player.arms.l.rotation.x = -cT * 2.2;
  player.arms.l.rotation.z = -cT * 0.5;
  player.group.userData.head.rotation.x = cT * 0.35;

  player.group.position.copy(player.pos);
  player.group.rotation.y = player.rot;
}

function updateCamera(dt){
  const back = new THREE.Vector3(Math.sin(player.rot), 0, Math.cos(player.rot));
  const dist = Math.max(5.2, Math.min(6.8, innerWidth/150));
  const bob = Math.sin(cameraBob) * (player.running ? 0.06 : 0.03);
  const desired = player.pos.clone()
    .addScaledVector(back, dist)
    .add(new THREE.Vector3(0, dist*0.55 + 1.3 + bob, 0));
  camera.position.lerp(desired, Math.min(1, dt*4));
  camTarget.copy(player.pos).add(new THREE.Vector3(0, 1.5, 0));
  camera.lookAt(camTarget);
}

function updateInteractPrompt(){
  if(state.dialogueOpen){ promptEl.classList.remove('show'); return; }
  let best = null, bestD = Infinity;
  for(const it of interactables){
    const d = Math.hypot(player.pos.x - it.x, player.pos.z - it.z);
    if(d < it.r && d < bestD){ best = it; bestD = d; }
  }
  currentInteract = best;
  if(best){ promptEl.textContent = '👉 ' + best.label + ' (دکمه E)'; promptEl.classList.add('show'); }
  else promptEl.classList.remove('show');
}

function updateSheep(dt){
  if(!state.sheepFollowing) return;
  sheep.forEach((s, i)=>{
    const target = player.pos.clone().add(new THREE.Vector3(
      Math.cos(i*1.1)*3.2, 0, Math.sin(i*1.1)*3.2));
    const dir = target.clone().sub(s.group.position);
    const d = dir.length();
    if(d > 0.5){
      dir.normalize();
      s.group.position.addScaledVector(dir, Math.min(d, 3.0*dt));
      s.group.rotation.y = Math.atan2(dir.x, dir.z) + Math.PI;
    }
  });

  const dxp = player.pos.x - 70, dzp = player.pos.z + 45;
  if(!state.sheepInPasture && dxp*dxp + dzp*dzp < 24*24){
    state.sheepInPasture = true;
    showToast('🐑 گله رسید به چراگاه!');
  }
  if(state.sheepInPasture && state.hour >= 19 && !state.waitingForNight){
    state.waitingForNight = true;
    showToast('🌙 غروب شد — گله رو برگردون!');
  }
  const dxb = player.pos.x - 60, dzb = player.pos.z + 10;
  if(state.waitingForNight && dxb*dxb + dzb*dzb < 6*6){
    state.mode = 'idle';
    state.sheepFollowing = false;
    state.sheepInPasture = false;
    state.waitingForNight = false;
    state.money += 700;
    showToast('✅ گله برگشت — +۷۰۰'); updateHUD();
  }
}

function updateLighting(){
  const h = state.hour + state.minute/60;
  let sunI = 0;
  if(h >= 6 && h < 8){ sunI = (h-6)/2 * 1.35; }
  else if(h >= 8 && h < 17){ sunI = 1.35; }
  else if(h >= 17 && h < 20){ sunI = 1.35 * (1 - (h-17)/3); }
  else { sunI = 0.06; }

  sun.intensity = sunI;
  ambientNight.intensity = 0.4 * (1 - Math.min(1, sunI/1.35));
  hemi.intensity = 0.15 + 0.5 * Math.min(1, sunI/1.35);

  // آسمان
  const dayTop = new THREE.Color(0x3a7ab8);
  const dayBot = new THREE.Color(0xcfd8e0);
  const sunsetTop = new THREE.Color(0x4a3a6a);
  const sunsetBot = new THREE.Color(0xff9966);
  const nightTop = new THREE.Color(0x0a0e1e);
  const nightBot = new THREE.Color(0x1a2040);

  let top, bot;
  if(h >= 5 && h < 7){
    const t = (h-5)/2;
    top = nightTop.clone().lerp(sunsetTop, t);
    bot = nightBot.clone().lerp(sunsetBot, t);
  } else if(h >= 7 && h < 17){
    const t = Math.min(1, (h-7)/3);
    top = sunsetTop.clone().lerp(dayTop, t);
    bot = sunsetBot.clone().lerp(dayBot, t);
  } else if(h >= 17 && h < 19){
    const t = (h-17)/2;
    top = dayTop.clone().lerp(sunsetTop, t);
    bot = dayBot.clone().lerp(sunsetBot, t);
  } else if(h >= 19 && h < 21){
    const t = (h-19)/2;
    top = sunsetTop.clone().lerp(nightTop, t);
    bot = sunsetBot.clone().lerp(nightBot, t);
  } else {
    top = nightTop; bot = nightBot;
  }
  skyMat.uniforms.topColor.value.copy(top);
  skyMat.uniforms.bottomColor.value.copy(bot);

  // نور خورشید
  const ang = ((h - 6) / 24) * Math.PI * 2;
  sun.position.set(Math.cos(ang)*80, Math.max(10, Math.sin(ang)*90), 40);

  // موقعیت خورشید بصری
  const R = 300;
  sunDisc.position.set(Math.cos(ang)*R*0.5, Math.sin(ang)*R*0.4, 200);
  sunDisc.lookAt(0, 20, 0);
  sunDisc.material.opacity = Math.max(0, Math.min(0.9, sunI/1.35));

  // فوگ
  scene.fog.color.copy(bot);
  const fogDensity = h >= 6 && h < 19 ? 0.0075 : 0.012;
  scene.fog.density = fogDensity;
}

let lastHudUpdate = 0;
function animate(){
  requestAnimationFrame(animate);
  const dt = Math.min(0.05, clock.getDelta());

  if(state.started && !state.paused){
    timeAccum += dt;
    while(timeAccum >= REAL_SEC_PER_GAME_MIN){
      timeAccum -= REAL_SEC_PER_GAME_MIN;
      advanceTime(1);
    }
    updatePlayer(dt);
    updateSheep(dt);
    updateDust(dt);
    updateCamera(dt);
    updateInteractPrompt();
    updateLighting();
    // موقعیت آسمان با دوربین
    sky.position.copy(camera.position);

    lastHudUpdate += dt;
    if(lastHudUpdate > 0.25){ lastHudUpdate = 0; updateHUD(); }
  }
  renderer.render(scene, camera);
}
animate();
